--- SQL ---
WITH __ad AS (
  SELECT
    ad_source_type,
    media_channel_nm,
    perf_date_sk,
    last_stmt_amt,
    ad_cost
  FROM GN_DW.GOLD.WIDE_AD_COMBINED
), __date AS (
  SELECT
    month,
    year,
    date_sk
  FROM GN_DW.GOLD.DIM_DATE
)
SELECT
  media AS "매체",
  cost AS "광고비(원)",
  settle AS "디지털 최종정산금액(원)",
  CAST(ROUND(share, 2) AS DECIMAL(38, 2)) AS "광고비 비중(%)",
  total_cost AS "전체 광고비(원)",
  total_settle AS "전체 디지털 최종정산금액(원)"
FROM (
  SELECT
    COALESCE(__ad.media_channel_nm, '(미매핑)') AS media,
    SUM(__ad.ad_cost) AS cost,
    SUM(__ad.last_stmt_amt) AS settle,
    SUM(__ad.ad_cost) / NULLIF(NULLIF(SUM(SUM(__ad.ad_cost)) OVER (), 0), 0) * 100 AS share,
    SUM(SUM(__ad.ad_cost)) OVER () AS total_cost,
    SUM(SUM(__ad.last_stmt_amt)) OVER () AS total_settle
  FROM __ad
  JOIN __date
    ON __ad.perf_date_sk = __date.date_sk
  WHERE
    __ad.ad_source_type = 'DIGITAL' AND __date.year = 2026 AND __date.month = 9
  GROUP BY
    1
)
ORDER BY
  cost DESC NULLS LAST /* Generated by Cortex */;
