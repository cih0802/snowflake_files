## 반영 후 검증 SQL (개발계 기대값)

```sql
-- ① 고위험 미매칭: 기대 = amt_unm 0 · pay_unm 소수(개발계 220/7,016)
SELECT r.CHURN_GRADE, COUNT(*) n, COUNT_IF(a.AMOUNT_BAND='(미매칭)') amt_unm,
       COUNT_IF(a.PAYMENT_METHOD_NAME='(미매칭)') pay_unm, COUNT_IF(a.ATTR_FALLBACK) fallback
FROM GN_DW.SERVING.ML_MEMBER_RISK_V r JOIN GN_DW.SERVING.ML_MEMBER_ATTR_ASOF_V a USING (STDR_MT, MBER_NO)
WHERE r.STDR_MT = (SELECT MAX(STDR_MT) FROM GN_DW.SERVING.ML_MEMBER_RISK_V) AND r.HAS_CHURN_PRED
GROUP BY 1 ORDER BY 1;

-- ② grain: 기대 = n = d
SELECT COUNT(*) n, COUNT(DISTINCT STDR_MT, MBER_NO) d FROM GN_DW.SERVING.ML_MEMBER_ATTR_ASOF_V;

-- ③ 배분 합계 보존: 기대 = 축 3개 모두 alloc ≈ orig
SELECT a.ALLOC_AXIS, SUM(a.ALLOC_FORECAST_AMT) alloc, ANY_VALUE(o.orig) orig
FROM GN_DW.SERVING.ML_FEE_FORECAST_ALLOC_V a
CROSS JOIN (SELECT SUM(FORECAST_AMT) orig FROM GN_DW.SERVING.ML_FEE_FORECAST_V
            WHERE STDR_MT = (SELECT MAX(STDR_MT) FROM GN_DW.SERVING.ML_FEE_FORECAST_V)) o
WHERE a.STDR_MT = (SELECT MAX(STDR_MT) FROM GN_DW.SERVING.ML_FEE_FORECAST_V)
GROUP BY 1;
```
